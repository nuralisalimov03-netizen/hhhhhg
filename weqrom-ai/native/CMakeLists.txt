cmake_minimum_required(VERSION 3.22)
project(weqrom_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED LLAMA_SOURCE_DIR)
    message(FATAL_ERROR "Pass the pinned llama.cpp b6500 checkout as LLAMA_SOURCE_DIR")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(flag LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_CURL GGML_OPENMP GGML_NATIVE GGML_CPU_KLEIDIAI GGML_BACKEND_DL GGML_LLAMAFILE)
    set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${LLAMA_SOURCE_DIR}" llama EXCLUDE_FROM_ALL)
add_library(weqrom_engine STATIC engine.cpp)
target_link_libraries(weqrom_engine PUBLIC llama)
target_include_directories(weqrom_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(ANDROID)
    add_library(weqrom_ai SHARED jni.cpp)
    target_link_libraries(weqrom_ai PRIVATE weqrom_engine)
    target_link_options(weqrom_ai PRIVATE "-Wl,-z,max-page-size=16384")
else()
    add_executable(weqrom-ai-run host.cpp)
    target_link_libraries(weqrom-ai-run PRIVATE weqrom_engine)
endif()
